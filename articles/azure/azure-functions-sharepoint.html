Azure Functions are the default back end for Microsoft 365 solutions: HTTP APIs for SPFx and Power Platform, timers that replace SharePoint timer jobs, and queue workers for webhooks and long-running tasks.

#.NET isolated + PnP Core SDK (certificate or MI)

Program.csC#
var host = new HostBuilder()
  .ConfigureFunctionsWorkerDefaults()
  .ConfigureServices((ctx, services) =>
  {
      services.AddApplicationInsightsTelemetryWorkerService();
      services.AddPnPCore(o =>
      {
          o.PnPContext.GraphFirst = true;
          o.Sites.Add("Quality", new PnPCoreSiteOptions { SiteUrl = ctx.Configuration["QualitySiteUrl"] });
      });
      services.AddPnPCoreAuthentication(a =>
      {
          a.Credentials.Configurations.Add("cert", new PnPCoreAuthenticationCredentialConfigurationOptions
          {
              ClientId = ctx.Configuration["ClientId"], TenantId = ctx.Configuration["TenantId"],
              X509Certificate = new PnPCoreAuthenticationX509CertificateOptions
                  { StoreName = "My", StoreLocation = "CurrentUser", Thumbprint = ctx.Configuration["CertThumbprint"] }
          });
          a.Credentials.DefaultConfiguration = "cert";
          a.Sites.Add("Quality", new PnPCoreAuthenticationSiteOptions { AuthenticationProviderName = "cert" });
      });
  })
  .Build();
host.Run();
Timer function: close stale drafts nightlyC#
public class CloseStaleDrafts(IPnPContextFactory factory, ILogger<CloseStaleDrafts> log)
{
    [Function("CloseStaleDrafts")]
    public async Task Run([TimerTrigger("0 30 1 * * *")] TimerInfo timer)          // 01:30 UTC daily
    {
        using var ctx = await factory.CreateAsync("Quality");
        var list = await ctx.Web.Lists.GetByTitleAsync("Deviations");
        string cutoff = DateTime.UtcNow.AddDays(-30).ToString("yyyy-MM-ddTHH:mm:ssZ");
        string caml = $@"<View><Query><Where><And>
                           <Eq><FieldRef Name='Status'/><Value Type='Choice'>Draft</Value></Eq>
                           <Lt><FieldRef Name='Modified'/><Value Type='DateTime' IncludeTimeValue='TRUE'>{cutoff}</Value></Lt>
                         </And></Where></Query><RowLimit Paged='TRUE'>500</RowLimit></View>";
        await list.LoadItemsByCamlQueryAsync(new CamlQueryOptions { ViewXml = caml, DatesInUtc = true });

        foreach (var item in list.Items.AsRequested())
        {
            item["Status"] = "Closed – inactive";
            await item.UpdateBatchAsync();                // batched
        }
        await ctx.ExecuteAsync();
        log.LogInformation("Closed {Count} stale drafts", list.Items.Length);
    }
}

#Node (v4 model) + Graph SDK with managed identity

src/functions/openDeviations.tsTypeScript
import { app, HttpRequest, HttpResponseInit } from '@azure/functions';
import { DefaultAzureCredential } from '@azure/identity';
import { Client } from '@microsoft/microsoft-graph-client';
import { TokenCredentialAuthenticationProvider } from '@microsoft/microsoft-graph-client/authProviders/azureTokenCredentials';

const graph = Client.initWithMiddleware({
  authProvider: new TokenCredentialAuthenticationProvider(new DefaultAzureCredential(), { scopes: ['https://graph.microsoft.com/.default'] })
});

app.http('openDeviations', {
  methods: ['GET'], authLevel: 'anonymous',            // protect with Easy Auth (Entra ID)
  handler: async (req: HttpRequest): Promise<HttpResponseInit> => {
    const res = await graph.api(`/sites/${process.env.SITE_ID}/lists/Deviations/items`)
      .header('Prefer', 'HonorNonIndexedQueriesWarningMayFailRandomly')
      .expand('fields($select=Title,Status,DueDate)')
      .filter("fields/Status eq 'Open'").top(100).get();
    return { jsonBody: res.value.map((i: any) => i.fields) };
  }
});
Long work goes to queues

HTTP-triggered functions should return quickly. For work that may take minutes (bulk updates, document generation), enqueue a message and process it with a queue-triggered function with retries and poison-queue handling.

#Interview questions

Q1How do Azure Functions replace SharePoint timer jobs?

A timer-triggered function (CRON schedule) running with an Entra app or managed identity (ideally Sites.Selected) performs the scheduled work through Graph or PnP Core SDK, with logging in Application Insights and no server infrastructure.

Q2Which SDK would you use to work with SharePoint lists from .NET in Azure?

PnP Core SDK (modern, batching, Graph-first with REST fallback) or the Microsoft Graph SDK for Graph-covered scenarios. CSOM via PnP Framework remains for legacy APIs not yet covered elsewhere.
