Provider-Hosted Add-in CRUD – Read, Create, Update & Delete Code Templates (MVC + CSOM)

This article gives you a working CRUD template for a provider-hosted add-in built with ASP.NET MVC (.NET Framework 4.x) and CSOM. It manages a host-web list called Deviations (Title, Area choice, Status choice, DueDate date, Owner person). Read how provider-hosted add-ins work first if the token flow is new to you.

#Request flow in the template

#1. The model

Models/Deviation.csC#
using System;
using System.ComponentModel.DataAnnotations;

public class Deviation
{
    public int Id { get; set; }

    [Required, StringLength(255, MinimumLength = 5)]
    public string Title { get; set; }

    [Required]
    public string Area { get; set; }            // Production | Warehouse | Lab

    public string Status { get; set; } = "Submitted";

    [DataType(DataType.Date)]
    public DateTime? DueDate { get; set; }

    public string OwnerEmail { get; set; }      // resolved to a SharePoint user
    public string OwnerName { get; set; }
    public DateTime Modified { get; set; }
}

#2. The repository (all CSOM CRUD in one place)

Data/DeviationRepository.csC#
using System;
using System.Collections.Generic;
using System.Linq;
using Microsoft.SharePoint.Client;

public class DeviationRepository
{
    private const string ListTitle = "Deviations";
    private readonly ClientContext _ctx;

    public DeviationRepository(ClientContext ctx) => _ctx = ctx;

    private List List => _ctx.Web.Lists.GetByTitle(ListTitle);

    // READ (paged) ---------------------------------------------------------
    public (IList<Deviation> Items, string NextPage) GetPage(string status, string pagingInfo, int pageSize = 30)
    {
        string where = string.IsNullOrEmpty(status) ? "" :
            $"<Where><Eq><FieldRef Name='Status'/><Value Type='Choice'>{System.Security.SecurityElement.Escape(status)}</Value></Eq></Where>";

        var q = new CamlQuery
        {
            ViewXml = $"<View><Query>{where}<OrderBy><FieldRef Name='ID' Ascending='FALSE'/></OrderBy></Query>" +
                      $"<ViewFields><FieldRef Name='Title'/><FieldRef Name='Area'/><FieldRef Name='Status'/>" +
                      $"<FieldRef Name='DueDate'/><FieldRef Name='Owner'/><FieldRef Name='Modified'/></ViewFields>" +
                      $"<RowLimit Paged='TRUE'>{pageSize}</RowLimit></View>",
            ListItemCollectionPosition = string.IsNullOrEmpty(pagingInfo) ? null
                : new ListItemCollectionPosition { PagingInfo = pagingInfo }
        };

        ListItemCollection items = List.GetItems(q);
        _ctx.Load(items, c => c.ListItemCollectionPosition);
        _ctx.Load(items);
        _ctx.ExecuteQuery();

        var result = items.Select(Map).ToList();
        return (result, items.ListItemCollectionPosition?.PagingInfo);
    }

    public Deviation Get(int id)
    {
        ListItem item = List.GetItemById(id);
        _ctx.Load(item);
        _ctx.ExecuteQuery();
        return Map(item);
    }

    // CREATE ---------------------------------------------------------------
    public int Create(Deviation d)
    {
        ListItem item = List.AddItem(new ListItemCreationInformation());
        Apply(item, d);
        item.Update();
        _ctx.Load(item, i => i.Id);
        _ctx.ExecuteQuery();
        return item.Id;
    }

    // UPDATE (with optimistic concurrency) ---------------------------------
    public void Update(Deviation d, DateTime expectedModified)
    {
        ListItem item = List.GetItemById(d.Id);
        _ctx.Load(item, i => i["Modified"]);
        _ctx.ExecuteQuery();

        var current = ((DateTime)item["Modified"]).ToUniversalTime();
        if (Math.Abs((current - expectedModified.ToUniversalTime()).TotalSeconds) > 1)
            throw new InvalidOperationException("This deviation was changed by someone else. Reload and try again.");

        Apply(item, d);
        item.Update();
        _ctx.ExecuteQuery();
    }

    // DELETE (to recycle bin) ------------------------------------------------
    public void Delete(int id)
    {
        List.GetItemById(id).Recycle();
        _ctx.ExecuteQuery();
    }

    // helpers -----------------------------------------------------------------
    private void Apply(ListItem item, Deviation d)
    {
        item["Title"] = d.Title;
        item["Area"] = d.Area;
        item["Status"] = d.Status;
        item["DueDate"] = d.DueDate;
        if (!string.IsNullOrWhiteSpace(d.OwnerEmail))
        {
            User owner = _ctx.Web.EnsureUser(d.OwnerEmail);   // resolves & adds to site if needed
            _ctx.Load(owner, u => u.Id);
            _ctx.ExecuteQuery();
            item["Owner"] = new FieldUserValue { LookupId = owner.Id };
        }
    }

    private static Deviation Map(ListItem i)
    {
        var owner = i["Owner"] as FieldUserValue;
        return new Deviation
        {
            Id = i.Id,
            Title = i["Title"] as string,
            Area = i["Area"] as string,
            Status = i["Status"] as string,
            DueDate = i["DueDate"] as DateTime?,
            OwnerName = owner?.LookupValue,
            OwnerEmail = owner?.Email,
            Modified = (DateTime)i["Modified"]
        };
    }
}

#3. The controller

Controllers/DeviationsController.csC#
using System;
using System.Web.Mvc;

[SharePointContextFilter]                 // ensures a valid SharePoint context/token
public class DeviationsController : Controller
{
    private ClientContext CreateContext()
    {
        var spContext = SharePointContextProvider.Current.GetSharePointContext(HttpContext);
        // User + add-in token for the HOST web
        return spContext.CreateUserClientContextForSPHost();
    }

    public ActionResult Index(string status, string page)
    {
        using (var ctx = CreateContext())
        {
            var (items, next) = new DeviationRepository(ctx).GetPage(status, page);
            ViewBag.NextPage = next;
            ViewBag.Status = status;
            return View(items);
        }
    }

    public ActionResult Create() => View(new Deviation());

    [HttpPost, ValidateAntiForgeryToken]
    public ActionResult Create(Deviation model)
    {
        if (!ModelState.IsValid) return View(model);
        using (var ctx = CreateContext())
        {
            try
            {
                int id = new DeviationRepository(ctx).Create(model);
                TempData["Message"] = $"Deviation #{id} created.";
                return RedirectToAction("Index", new { SPHostUrl = Request.QueryString["SPHostUrl"] });
            }
            catch (ServerException ex)   // SharePoint-side error (permissions, validation, threshold)
            {
                ModelState.AddModelError("", $"SharePoint error: {ex.Message}");
                return View(model);
            }
        }
    }

    public ActionResult Edit(int id)
    {
        using (var ctx = CreateContext())
            return View(new DeviationRepository(ctx).Get(id));
    }

    [HttpPost, ValidateAntiForgeryToken]
    public ActionResult Edit(Deviation model, DateTime originalModified)
    {
        if (!ModelState.IsValid) return View(model);
        using (var ctx = CreateContext())
        {
            try
            {
                new DeviationRepository(ctx).Update(model, originalModified);
                return RedirectToAction("Index", new { SPHostUrl = Request.QueryString["SPHostUrl"] });
            }
            catch (InvalidOperationException conflict)
            {
                ModelState.AddModelError("", conflict.Message);
                return View(model);
            }
        }
    }

    [HttpPost, ValidateAntiForgeryToken]
    public ActionResult Delete(int id)
    {
        using (var ctx = CreateContext())
            new DeviationRepository(ctx).Delete(id);
        return RedirectToAction("Index", new { SPHostUrl = Request.QueryString["SPHostUrl"] });
    }
}
Always carry SPHostUrl

SharePointContextFilter needs SPHostUrl (and friends) on every request to find the cached context. Generated templates add it through ?{StandardTokens} on the start page. Your redirects and links must keep passing it (or store it in session/cookie). Missing it is the most common "redirect loop" bug.

#4. A Razor view (Index)

Views/Deviations/Index.cshtmlHTML
@model IList<Deviation>
@{ var host = Request.QueryString["SPHostUrl"]; }
<h2>Deviations</h2>
@if (TempData["Message"] != null) { <div class="alert alert-success">@TempData["Message"]</div> }
<p>@Html.ActionLink("New deviation", "Create", new { SPHostUrl = host }, new { @class = "btn btn-primary" })</p>
<table class="table">
  <thead><tr><th>ID</th><th>Title</th><th>Area</th><th>Status</th><th>Due</th><th>Owner</th><th></th></tr></thead>
  <tbody>
  @foreach (var d in Model) {
    <tr>
      <td>@d.Id</td><td>@d.Title</td><td>@d.Area</td><td>@d.Status</td>
      <td>@(d.DueDate?.ToString("dd MMM yyyy"))</td><td>@d.OwnerName</td>
      <td>
        @Html.ActionLink("Edit", "Edit", new { id = d.Id, SPHostUrl = host })
        @using (Html.BeginForm("Delete", "Deviations", new { id = d.Id, SPHostUrl = host }, FormMethod.Post, new { style = "display:inline" })) {
          @Html.AntiForgeryToken()
          <button type="submit" class="btn btn-link text-danger" onclick="return confirm('Delete?')">Delete</button>
        }
      </td>
    </tr>
  }
  </tbody>
</table>
@if (ViewBag.NextPage != null) {
  @Html.ActionLink("Next page →", "Index", new { page = ViewBag.NextPage, status = ViewBag.Status, SPHostUrl = host })
}

#5. The same CRUD with REST from the remote web

When you prefer REST (for example from a Web API controller), take the access token from the SharePoint context:

REST with HttpClient and the add-in access tokenC#
var spContext = SharePointContextProvider.Current.GetSharePointContext(HttpContext);
string token = spContext.UserAccessTokenForSPHost;
string host = spContext.SPHostUrl.ToString().TrimEnd('/');

using (var http = new HttpClient())
{
    http.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", token);
    http.DefaultRequestHeaders.Accept.ParseAdd("application/json;odata=nometadata");

    // READ
    var json = await http.GetStringAsync($"{host}/_api/web/lists/getbytitle('Deviations')/items?$select=Id,Title,Status&$top=50");

    // CREATE (no X-RequestDigest needed with a bearer token)
    var body = new StringContent("{\"Title\":\"Created via REST\",\"Area\":\"Lab\"}", Encoding.UTF8, "application/json");
    body.Headers.ContentType.Parameters.Add(new NameValueHeaderValue("odata", "nometadata"));
    var created = await http.PostAsync($"{host}/_api/web/lists/getbytitle('Deviations')/items", body);

    // UPDATE
    var upd = new HttpRequestMessage(HttpMethod.Post, $"{host}/_api/web/lists/getbytitle('Deviations')/items(42)")
    { Content = new StringContent("{\"Status\":\"Closed\"}", Encoding.UTF8, "application/json") };
    upd.Headers.Add("X-HTTP-Method", "MERGE");
    upd.Headers.Add("IF-MATCH", "*");
    await http.SendAsync(upd);

    // DELETE
    var del = new HttpRequestMessage(HttpMethod.Post, $"{host}/_api/web/lists/getbytitle('Deviations')/items(42)");
    del.Headers.Add("X-HTTP-Method", "DELETE");
    del.Headers.Add("IF-MATCH", "*");
    await http.SendAsync(del);
}

#6. App-only context (no user), e.g. for a nightly job

App-only ClientContext (low-trust) and high-trust equivalentsC#
// LOW-TRUST (ACS): requires AllowAppOnlyPolicy="true" + ClientId/ClientSecret in config
Uri site = new Uri("https://contoso.sharepoint.com/sites/quality");
string realm = TokenHelper.GetRealmFromTargetUrl(site);
string appOnlyToken = TokenHelper.GetAppOnlyAccessToken(TokenHelper.SharePointPrincipal, site.Authority, realm).AccessToken;
using (var ctx = TokenHelper.GetClientContextWithAccessToken(site.ToString(), appOnlyToken))
{
    new DeviationRepository(ctx).GetPage(null, null);
}

// HIGH-TRUST (on-prem S2S): token signed with the certificate, no user identity
using (var ctx = TokenHelper.GetS2SClientContextWithWindowsIdentity(site, null))   // null = app-only
{
    /* ... */
}
SharePoint Online in 2026

The low-trust (ACS) code above no longer works in SharePoint Online since ACS retired on 2 April 2026. For SPO, use an Entra ID app registration with a certificate and Microsoft Graph or SharePoint permissions (ideally Sites.Selected), and call CSOM through PnP Framework / PnP Core SDK with MSAL tokens. See Provider-hosted challenges & migration and Sites.Selected.

#Testing checklist

  • Install in a test site as a non-admin user; verify the permission prompt matches the manifest.
  • CRUD as a user with Contribute (works) and Read (should fail gracefully).
  • Item with an ampersand and quotes in Title (CAML escaping).
  • List with more than 5,000 items (paging and indexed filter on Status).
  • Two browsers editing the same item (concurrency message).
  • Session timeout or app pool recycle (context re-acquired, no redirect loop).
  • Remote web certificate renewal (high-trust): update the trusted issuer cert before expiry.

#Interview questions

Q1How does a provider-hosted MVC controller get a ClientContext for the host web?

Decorate the controller with [SharePointContextFilter], then call SharePointContextProvider.Current.GetSharePointContext(HttpContext) and CreateUserClientContextForSPHost() for user+app, or CreateAppOnlyClientContextForSPHost() for app-only. The provider caches context and access tokens in session.

Q2Why don't REST calls from a provider-hosted add-in need X-RequestDigest?

The digest protects cookie-based authentication against CSRF. Add-ins authenticate with an OAuth bearer token in the Authorization header, so no form digest is required.

Q3How do you implement paging over a large list in CSOM?

Use RowLimit Paged='TRUE' with an indexed sort (e.g. ID), read ListItemCollectionPosition.PagingInfo from the result, and pass it back as CamlQuery.ListItemCollectionPosition on the next request. Store it in the "next page" link.
