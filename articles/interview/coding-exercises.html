#SPFx

Q5Write a React hook that loads items and cancels state updates on unmount.
TypeScriptTypeScript
export function useItems(service: { get(): Promise<any[]> }) {
  const [items, setItems] = React.useState<any[]>([]);
  const [error, setError] = React.useState<string>();
  React.useEffect(() => {
    let alive = true;
    service.get().then(r => alive && setItems(r)).catch(e => alive && setError(e.message));
    return () => { alive = false; };
  }, [service]);
  return { items, error };
}

#PowerShell

Q10PnP PowerShell: report lists with more than 5,000 items across all sites.
PowerShellPowerShell
Connect-PnPOnline -Url https://contoso-admin.sharepoint.com -Interactive -ClientId $id
Get-PnPTenantSite | ForEach-Object {
  Connect-PnPOnline -Url $_.Url -Interactive -ClientId $id
  Get-PnPList | Where-Object { -not $_.Hidden -and $_.ItemCount -gt 5000 } |
    Select-Object @{n='Site';e={$_.ParentWebUrl}}, Title, ItemCount
} | Export-Csv large-lists.csv -NoTypeInformation
